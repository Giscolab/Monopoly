# --- Development sanitizers option: enable Address/Undefined sanitizers easily ---
# Add this option to help catch memory errors during refactoring. Disabled by
# default because sanitizers change build/link flags and are intended for
# developer/debug builds (Linux/macOS with clang/gcc). On Windows prefer
# clang/clang-cl if you need sanitizers; MSVC support is limited.
option(ENABLE_SANITIZERS "Enable Address/Undefined/Leak sanitizers for development builds" OFF)

if(ENABLE_SANITIZERS)
    if(MSVC)
        message(WARNING "Sanitizers are not fully supported on MSVC in this script; consider using Clang on Windows or use VS AddressSanitizer support.")
    else()
        # If no build type was set, prefer Debug for sanitizer builds so we have
        # better diagnostics. Sanitizers work best with low optimization.
        if(NOT CMAKE_BUILD_TYPE)
            set(CMAKE_BUILD_TYPE Debug CACHE STRING "" FORCE)
        endif()

        set(SANITIZE_FLAGS "-fsanitize=address,undefined -fno-omit-frame-pointer")
        # Use -O1 for sanitizers to reduce false positives and preserve debuggability
        add_compile_options("$<$<CONFIG:Debug>:-O1>" ${SANITIZE_FLAGS} -g)
        add_link_options(${SANITIZE_FLAGS})
    endif()
endif()

if(MSVC)
    # CMake's MSVC probe can decode the French UTF-8 nonbreaking spaces as
    # OEM text. Ninja then records zero header dependencies and retains stale
    # ABI objects. Repair only this detected corruption; other locales retain
    # their compiler-probed prefix (an English language pack is not required).
    if(CMAKE_GENERATOR MATCHES "Ninja")
        string(REPLACE "┬á" " " CMAKE_CL_SHOWINCLUDES_PREFIX
            "${CMAKE_CL_SHOWINCLUDES_PREFIX}")
    endif()
    # MSVC does not enable standard C++ stack unwinding for these Ninja
    # targets automatically in the current VS 18 toolchain.
    add_compile_options(
        $<$<COMPILE_LANGUAGE:CXX>:/EHsc>
    )
endif()

# Keep the shared shader directory for GPU consumers and test targets.
# The application also stages these files next to its configuration-specific
# executable below (build/Debug or build/Release for Visual Studio).
# configure_file registers shader changes as build dependencies.
set(MONOPOLY_RUNTIME_SHADER_FILES)
# The checked-in SPIR-V fragment still describes the pre-texture contract;
# do not package it until it can be regenerated by a SPIR-V-capable compiler.
foreach(WORLD_SHADER IN ITEMS World3D.vert.dxil World3D.frag.dxil
    World3D.vert.msl World3D.frag.msl
    ModernPBR.vert.dxil ModernPBR.frag.dxil
    ModernPBR.vert.msl ModernPBR.frag.msl
    ModernPBR.vert.spv ModernPBR.frag.spv)
    configure_file("shaders/generated/${WORLD_SHADER}"
        "shaders/${WORLD_SHADER}" COPYONLY)
    list(APPEND MONOPOLY_RUNTIME_SHADER_FILES
        "${CMAKE_CURRENT_SOURCE_DIR}/shaders/generated/${WORLD_SHADER}")
endforeach()
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    # Explicit developer target; ordinary builds use checked-in generated assets.
    # Tool binaries and regenerated outputs stay local to this build directory.
    add_custom_target(MonopolyCompileShaders
        COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tools/compile_world3d_shaders.py"
            --bootstrap --name ModernPBR
            --cmake "${CMAKE_COMMAND}"
            --cache "${CMAKE_CURRENT_BINARY_DIR}/shader-tools"
            --output "${CMAKE_CURRENT_BINARY_DIR}/compiled-shaders"
        COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tools/compile_world3d_shaders.py"
            --name World3D
            --cache "${CMAKE_CURRENT_BINARY_DIR}/shader-tools"
            --output "${CMAKE_CURRENT_BINARY_DIR}/compiled-shaders"
        COMMENT "Compiling verified DXIL, SPIR-V and MSL with project-local pinned tools"
        VERBATIM
    )
endif()

# SDL_GetBasePath() resolves to the executable directory, not the build
# root or working directory. Always refresh runtime shaders, including when
# only a shader changed or a staged file was removed without relinking C++.
# TARGET_FILE_DIR adds no target dependency with CMP0112 NEW (CMake >= 3.19).
add_custom_target(MonopolyRuntimeShaders
    COMMAND ${CMAKE_COMMAND} -E make_directory
        "$<TARGET_FILE_DIR:MonopolyModern>/shaders"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        ${MONOPOLY_RUNTIME_SHADER_FILES}
