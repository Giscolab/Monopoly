# --- Development sanitizers option: enable Address/Undefined sanitizers easily ---
# Add this option to help catch memory errors during refactoring. Disabled by
# default because sanitizers change build/link flags and are intended for
# developer/debug builds (Linux/macOS with clang/gcc). On Windows prefer
# clang/clang-cl if you need sanitizers; MSVC support is limited.
option(ENABLE_SANITIZERS "Enable Address/Undefined/Leak sanitizers for development builds" OFF)

if(ENABLE_SANITIZERS)
    if(MSVC)
        message(WARNING "Sanitizers are not fully supported on MSVC in this script; consider using Clang on Windows or use VS AddressSanitizer support.")
    else()
        # If no build type was set, prefer Debug for sanitizer builds so we have
        # better diagnostics. Sanitizers work best with low optimization.
        if(NOT CMAKE_BUILD_TYPE)
            set(CMAKE_BUILD_TYPE Debug CACHE STRING "" FORCE)
        endif()

        set(SANITIZE_FLAGS "-fsanitize=address,undefined -fno-omit-frame-pointer")
        # Use -O1 for sanitizers to reduce false positives and preserve debuggability
        add_compile_options("$<$<CONFIG:Debug>:-O1>" ${SANITIZE_FLAGS} -g)
        add_link_options(${SANITIZE_FLAGS})
    endif()
endif()

if(MSVC)
    # CMake's MSVC probe can decode the French UTF-8 nonbreaking spaces as
    # OEM text. Ninja then records zero header dependencies and retains stale
    # ABI objects. Repair only this detected corruption; other locales retain
    # their compiler-probed prefix (an English language pack is not required).
    if(CMAKE_GENERATOR MATCHES "Ninja")
        string(REPLACE "┬á" " " CMAKE_CL_SHOWINCLUDES_PREFIX
            "${CMAKE_CL_SHOWINCLUDES_PREFIX}")
    endif()
    # MSVC does not enable standard C++ stack unwinding for these Ninja
    # targets automatically in the current VS 18 toolchain.
    add_compile_options(
        $<$<COMPILE_LANGUAGE:CXX>:/EHsc>
    )
endif()

# Keep generated shader assets beside the application, independent of cwd.
# configure_file also registers shader changes as build dependencies.
# The checked-in SPIR-V fragment still describes the pre-texture contract;
# do not package it until it can be regenerated by a SPIR-V-capable compiler.
foreach(WORLD_SHADER IN ITEMS World3D.vert.dxil World3D.frag.dxil
    World3D.vert.msl World3D.frag.msl)
    configure_file("shaders/generated/${WORLD_SHADER}"
        "shaders/${WORLD_SHADER}" COPYONLY)
endforeach()
file(REMOVE
    "${CMAKE_CURRENT_BINARY_DIR}/shaders/World3D.vert.spv"
    "${CMAKE_CURRENT_BINARY_DIR}/shaders/World3D.frag.spv")

# Source/ZLib/ZImplode.c proves that the historical implode/explode names
# wrap ordinary zlib streams (windowBits = 15). Prefer an installed package;
# otherwise build the pinned upstream release, never the private SDL copy.
find_package(ZLIB 1.2.9 QUIET)

if(ZLIB_FOUND)
    set(MONOPOLY_ZLIB_TARGET ZLIB::ZLIB)
else()
    set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
    set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
    set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
    set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)

    FetchContent_Declare(
        zlib
        GIT_REPOSITORY https://github.com/madler/zlib.git
        GIT_TAG v1.3.2
        GIT_SHALLOW TRUE
    )

    FetchContent_MakeAvailable(zlib)
    set(MONOPOLY_ZLIB_TARGET ZLIB::ZLIBSTATIC)
endif()
